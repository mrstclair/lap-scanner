async function toggleCam(){ ensureAudio(); const btn=document.getElementById('camBtn'); if(camOn){scanner.stop().then(()=>{camOn=false;btn.textContent='📷 Start camera';});return;} if(!navigator.mediaDevices||!navigator.mediaDevices.getUserMedia){setFeedback('err','Camera unsupported','Use the main page manual box.');return;} // Try rear camera first, fall back to any available camera (laptops have only front) let stream; try { stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } }); } catch (e1) { try { stream = await navigator.mediaDevices.getUserMedia({ video: true }); // any camera } catch (e2) { setFeedback('err','Camera: '+e2.name, e2.name==='NotFoundError'?'No camera detected on this device.':e2.message); return; } } stream.getTracks().forEach(t=>t.stop()); scanner=scanner||new Html5Qrcode('reader'); // Pick a camera from the list; prefer one labeled "back", else the first Html5Qrcode.getCameras().then(cams=>{ if(!cams||!cams.length){setFeedback('err','No camera found','This device has no usable camera.');return;} const back=cams.find(c=>/back|rear|environment/i.test(c.label)); const camId=(back||cams[0]).id; scanner.start(camId,{fps:10,qrbox:250},onScan,onScanError) .then(()=>{camOn=true;btn.textContent='⏹ Stop camera';}) .catch(e=>setFeedback('err','Camera start failed',String(e))); }).catch(e=>setFeedback('err','Camera list failed',String(e))); }
